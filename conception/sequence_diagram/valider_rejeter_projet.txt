@startuml UC_Valider_Publier_Projet

' ============================
'  Normes FSEG (ECB)
'  - Acteurs : Expert en finance, Administrateur
'  - <<boundary>> : IHM_Plateforme
'  - <<control>>  : Controleur_Projet
'  - <<entity>>   : Projet, Notification
' ============================

hide footbox
skinparam shadowing false
skinparam defaultFontName "Times New Roman"
skinparam defaultFontSize 12
skinparam sequenceMessageAlign left
skinparam maxMessageSize 45
skinparam wrapWidth 150
skinparam dpi 150
scale 0.85

title UC Valider (Expert) et Publier (Admin) un projet — Cycle de revue (ECB/FSEG)

actor "Expert en finance" as Exp
actor "Administrateur" as Adm
boundary "IHM_Plateforme" as IHM
control "Controleur_Projet" as CP
entity "Projet" as P
entity "Notification" as N

== Étape 1 : Validation de l'analyse IA par l'Expert ==
Exp -> IHM : validerAnalyseProjet(projectId, decision, feedback)
IHM -> CP : validerAnalyseProjet(expertId, projectId, decision, feedback)

CP -> P : charger(projectId)
alt Projet introuvable ou état invalide (status != UNDER_REVIEW)
  CP --> IHM : erreur("Action impossible dans cet état")
  IHM --> Exp : message d’erreur
else Décision autorisée
  alt decision = APPROVED
    CP -> P : maj(status=APPROVED, feedbackExpert=feedback)
    CP -> N : creer(type=PROJECT_APPROVED, destinataire=créateur)
    CP --> IHM : confirmation("Projet approuvé par l'expert")
  else decision = REJECTED
    CP -> P : maj(status=REJECTED, rejectionReason=feedback)
    CP -> N : creer(type=PROJECT_REJECTED, destinataire=créateur)
    CP --> IHM : confirmation("Projet rejeté")
  end
end

== Étape 2 : Publication finale par l'Administrateur ==
opt Si le projet a été approuvé (status = APPROVED)
  Adm -> IHM : publierProjet(projectId)
  IHM -> CP : publierProjet(adminId, projectId)

  CP -> P : charger(projectId)
  alt État invalide (status != APPROVED)
    CP --> IHM : erreur("Publication impossible : le projet doit être APPROVED")
    IHM --> Adm : message d’erreur
  else Publication autorisée
    CP -> P : maj(status=ACTIVE, publishedAt)
    CP -> N : creer(type=PROJECT_PUBLISHED, destinataire=créateur)
    CP --> IHM : confirmation("Projet publié et actif")
  end
end

@enduml
