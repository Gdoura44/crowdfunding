title UC Investir — Demande et confirmation d'investissement (ECB/FSEG)

actor "Utilisateur\n(investisseur)" as U
boundary "IHM_Contribution" as IHM
control "Controleur_Investissement" as CI
entity "Projet" as P
entity "Investissement" as Inv
entity "Transaction" as Tx
entity "Notification" as N
entity "Facture (Invoice)" as Invoice
entity "ExpertConsultation" as EC
participant "Passerelle de paiement" as Pay

== Démarrage de l’investissement ==
U -> IHM : investir(montant, idProjet, wantsConsultation, tipAmount)
IHM -> CI : initialiserInvestissement(investorId, projectId, amount, wantsConsultation, tipAmount)

CI -> P : charger(projectId)
CI -> P : verifierInvestissable(ACTIVE, !archive, dates OK, plafond OK)
alt Projet non investissable\n(inactif / expiré / plafond atteint)
  CI --> IHM : erreur("Projet non disponible")
  IHM --> U : message d’erreur
else Investisseur = créateur
  CI --> IHM : erreur("Interdit : auto-investissement")
  IHM --> U : message d’erreur
else Montant invalide
  CI --> IHM : erreur("Montant minimum 100 TND")
  IHM --> U : message d’erreur
else wantsConsultation = true et montant < 30% de l’objectif
  CI --> IHM : erreur("Consultation réservée aux contributions >= 30%")
  IHM --> U : message d’erreur
else Tentative d’investissement valide
  CI -> Pay : genererLienPaiement(amount, referenceId=investmentId)
  Pay --> CI : paymentUrl, providerPaymentId, provider
  CI -> Inv : creer(_id, investorId, projectId, amount, status=INITIATED, wantsConsultation, tipAmount)
  CI -> Tx : creer(investmentId, provider, providerPaymentId, amount, status=PENDING)
  CI --> IHM : retourner(paymentUrl)
  IHM --> U : redirectionPaiement(paymentUrl)
end

== Confirmation de paiement (Notification asynchrone) ==
Pay -> CI : notifierStatutPaiement(providerPaymentId, status, paymentMethod)
CI -> CI : verifierProvenanceRequete()
CI -> Tx : rechercher(providerPaymentId)
CI -> Inv : charger(tx.investmentId)
CI -> P : charger(inv.projectId)

alt Requête déjà traitée\n(Tx.status != PENDING)
  CI --> Pay : 200 OK (déjà traité)
else Traitement du paiement
  alt status = ECHEC
    CI -> Tx : maj(status=FAILED, paymentMethod)
    CI -> Inv : maj(status=FAILED)
    CI -> N : creer(userId=investorId, type=PAYMENT_FAILED, relatedEntity=INVESTMENT)
    CI --> Pay : 200 OK
  else status = SUCCES
    alt requestConsultation = true
      CI -> Tx : maj(status=SUCCEEDED, succeededAt, paymentMethod)
      CI -> Inv : maj(status=PENDING_CONSULTATION)
      CI -> Invoice : creerFacture(type=INVESTMENT)
      CI -> EC : creer(projectId, investorId, investmentId, status=OPEN)
      CI -> N : creer(type=PAYMENT_HELD_CONSULTATION, destinataire=investisseur)
      CI -> N : creer(type=NEW_ASSIGNMENT, destinataire=experts)
      CI --> Pay : 200 OK
    else requestConsultation = false (nominal)
      group Détection sur-financement (atomique)
        CI -> P : incrementer currentFunding si currentFunding+amount <= fundingGoal
      end

      alt Sur-financement détecté\n(incrément refusé)
        CI -> Pay : rembourserTransaction(providerPaymentId, amount)
        Pay --> CI : ok / échec
        alt remboursement OK
          CI -> Tx : maj(status=REFUNDED, refundStatus=SUCCEEDED, refundedAt)
          CI -> Inv : maj(status=REFUNDED)
          CI -> N : creer(type=OVERFUND_REFUNDED, relatedEntity=PROJECT)
        else remboursement KO
          CI -> Tx : maj(status=SUCCEEDED, refundStatus=FAILED, succeededAt)
          CI -> Inv : maj(status=SUCCESS)
          CI -> N : creer(type=REFUND_FAILED, relatedEntity=PROJECT)
        end
        CI --> Pay : 200 OK
      else Paiement confirmé et montant accepté
        CI -> Tx : maj(status=SUCCEEDED, succeededAt, paymentMethod)
        CI -> Inv : maj(status=SUCCESS)
        CI -> Invoice : creerFacture(type=INVESTMENT, investmentId)
        CI -> N : creer(type=PAYMENT_SUCCESS, relatedEntity=INVESTMENT)
        CI --> Pay : 200 OK
      end
    end
  end
end
